# 插件后端开发指南

> 如何编写 `entry.py`，与现有 `MiniGameController`、`MiniGameIntentService`、`SessionService` 完全对齐。
> 最后更新：2026-09-21

---

## 一、后端职责

```
用户操作（前端 entry.js）
    ↓ dispatchAction / miniGameInventoryAction
后端 API（miniGameInventoryAction.ts）
    ↓
MiniGameController.handleMiniGameTurn()
    ↓
entry.py · apply_action()  ← 插件在这里写游戏逻辑
    ↓
writeback 合并到 session.state
    ↓
AI 编排继续（NarrativeOrchestrator）
```

**后端做什么：**
1. 初始化小游戏（`setup_game`）：写入 `session.state.miniGame`
2. 处理用户动作（`apply_action`）：更新 `public_state`、`writeback`
3. 提供 AI 提示词（`get_trigger_prompt` / `get_intent_prompt`）：让 AI 理解游戏
4. 发放奖励（`settle_game`）：更新 `player.parameterCardJson`、`inventory`、`vars`

**后端不做什么（前端负责）：**
- 渲染 Canvas / 面板 UI
- 键盘/触摸输入采集
- 实时动画

# entry.py — 野外生存插件后端（完整注释参考）

def as_record(value: Any) -> Dict:
    """对齐 MiniGameController.ts 的 asRecord()"""
    if isinstance(value, dict):
        return value
    if hasattr(value, "__dict__"):
        return vars(value)
    return {}


def as_array(value: Any) -> List:
    """对齐 MiniGameController.ts 的 asArray()"""
    if isinstance(value, list):
        return value
    return []


def scalar_text(value: Any, fallback: str = "") -> str:
    """对齐 MiniGameController.ts 的 scalarText()"""
    if isinstance(value, str):
        return value.strip()
    if value is None:
        return fallback
    return str(value)


def unique_texts(items: List[str]) -> List[str]:
    """去重 + 过滤空字符串，对齐 gameEngine.ts"""
    seen = set()
    result = []
    for item in items:
        t = scalar_text(item)
        if t and t not in seen:
            seen.add(t)
            result.append(t)
    return result

def setup_game(
    state: Dict,
    session_id: str,
    player_id: str,
    npc_ids: List[str],
    difficulty: str = "normal",
    entry_source: str = "user_action",
    ctx: Optional[Dict] = None,
) -> Dict:
    """
    初始化小游戏状态，写入 session.state.miniGame。
    对齐 MiniGameController.ts 的 setup() 函数。

    参数：
        state       — 当前会话状态（session.state）
        session_id  — 会话 ID
        player_id   — 玩家角色 ID
        npc_ids     — NPC 角色 ID 列表
        difficulty  — 难度：normal / hard / nightmare
        entry_source— 入口来源：user_action / ai_trigger / panel_trigger
        ctx         — 可选的额外上下文

    返回：
        一个 dict，必须包含以下键（对齐 MiniGameController 的返回值）：
        {
            "ok": True,
            "mini_game_root": { ... },   # 写入 state.miniGame 的根对象
            "init_narration": "...",     # 开场叙事文本（AI 自动播报）
            "mentor_speech": "...",      # 陪练角色台词（可选）
            "pending_plan": { ... }      # 编排计划（可选）
        }
    """
    # 难度配置（对齐 entry.js 的 difficulties）
    diff_configs = {
        "normal":   {"wave_count": 5,  "enemy_scale": 1.0, "enemy_base_hp": 30},
        "hard":     {"wave_count": 8,  "enemy_scale": 1.5, "enemy_base_hp": 50},
        "nightmare":{"wave_count": 10, "enemy_scale": 2.0, "enemy_base_hp": 80},
    }
    config = diff_configs.get(difficulty, diff_configs["normal"])

    # 读取玩家参数卡（对齐 MiniGameController.ts 的 createPlayerParameterCard）
    player = as_record(state.get("player", {}))
    card = as_record(player.get("parameterCardJson", {}))
    player_level = max(1, int(float(card.get("level", 1))))

    # 读取 NPC 参数
    npcs = as_record(state.get("npcs", {}))
    ally_list = []
    for npc_id in npc_ids:
        npc = as_record(npcs.get(npc_id, {}))
        if npc.get("name"):
            ally_list.append({
                "id": npc_id,
                "name": scalar_text(npc.get("name")),
                "hp": int(float(npc.get("parameterCardJson", {}).get("hp", 50))),
                "max_hp": int(float(npc.get("parameterCardJson", {}).get("max_hp", 50))),
                "atk": int(float(npc.get("parameterCardJson", {}).get("atk", 10))),
            })

    # 构建 public_state（对齐 buildMiniGameUiStateItems 的数据来源）
    public_state = {
        "difficulty": difficulty,
        "wave_count": config["wave_count"],
        "current_wave": 0,
        "enemy_scale": config["enemy_scale"],
        "enemy_base_hp": config["enemy_base_hp"],
        "player_hp": int(float(card.get("hp", 100))),
        "player_max_hp": int(float(card.get("max_hp", 100))),
        "player_mp": int(float(card.get("mp", 50))),
        "player_max_mp": int(float(card.get("max_mp", 50))),
        "player_level": player_level,
        "player_atk": int(float(card.get("atk", 15))),
        "player_def": int(float(card.get("def", 5))),
        "player_money": int(float(card.get("money", 0))),
        "enemy_list": [],
        "ally_list": ally_list,
        "drops": [],
        "log": [],
        "current_status": "ready",
        "last_result": "",
        "total_damage_dealt": 0,
        "total_enemies_killed": 0,
        "active_enemy_count": 0,
    }

    # 构建完整的 miniGame 根对象（对齐 MiniGameController 的 root 结构）
    mini_game_root = {
        "session": {
            "game_type": GAME_TYPE,
            "status": STATUS_PREPARING,
            "phase": "ready",
            "round": 0,
            "entry_source": entry_source,
            "player_id": player_id,
            "npc_ids": npc_ids,
            "difficulty": difficulty,
            "public_state": public_state,
        },
        "ui": {
            "phase_label": "准备阶段",
            "narration": "",
            "rule_summary": f"野外生存挑战：{config['wave_count']}波怪物，击败所有敌人即可获得奖励。",
            "state_items": _build_state_items(public_state),
            "input_hint": "输入指令或使用技能按钮进行操作",
            "accepts_text_input": True,
        },
        "rulebook": {
            "displayName": DISPLAY_NAME,
            "version": "1.0.0",
            "goal": "击败所有波次的敌人，生存并获取奖励",
            "phaseOrder": ["ready", "battle", "wave_clear", "reward", "finish"],
            "ruleSummary": "野外生存战斗：每波敌人击败后进入短暂准备期，可升级或使用物品",
        },
        "actionLog": [],
        "writeback": {},
        "memorySummary": "",
        "passiveReentrySuppressed": False,
    }

    init_narration = (
        f"你踏入危机四伏的荒野，{config['wave_count']}波怪物正在前方等待着你的挑战。"
        f"第1波敌人即将出现，准备好了吗？"
    )

    return {
        "ok": True,
        "mini_game_root": mini_game_root,
        "init_narration": init_narration,
    }


def apply_action(
    state: Dict,
    action_id: str,
    params: Dict,
    ctx: Optional[Dict] = None,
) -> Dict:
    """
    处理用户动作，更新 public_state，返回 MiniGameStepResult。
    对齐 MiniGameController.ts 的 applyAction() 函数签名和返回值。

    参数：
        state     — 当前会话状态（session.state）
        action_id — 动作 ID（来自 entry.js dispatchAction）
        params    — 动作参数（如 {target: "enemy_0", skill: "fire_ball"}）
        ctx       — 额外上下文

    返回（对齐 MiniGameStepResult）：
        {
            "ok": True,
            "step_result": {
                "messages": [...],           # 可选：额外的消息列表
                "pending_narrative_plan": {  # 播报编排计划
                    "role": "narrator",
                    "roleType": "narrator",
                    "content": "...",
                },
                "reward_summary": {...},     # 奖励摘要（settle 时发放）
                "writeback": {...},          # 直接写入 state 的变更
                "mentor_speech": "...",      # 陪练台词
                "session_update": {...},     # 更新 miniGame.session
                "ui_update": {...},          # 更新 miniGame.ui
            }
        }
    """
    # 读取当前小游戏状态
    root = as_record(state.get("miniGame", {}))
    session = as_record(root.get("session", {}))
    public_state = as_record(session.get("public_state", {}))
    ui = as_record(root.get("ui", {}))
    player = as_record(state.get("player", {}))
    card = as_record(player.get("parameterCardJson", {}))

    result_messages = []
    pending_narration = ""
    reward_summary = {}
    writeback_delta = {}
    session_update = {}
    ui_update = {}

    action_id = str(action_id or "").strip()
    current_wave = int(float(public_state.get("current_wave", 0)))
    wave_count = int(float(public_state.get("wave_count", 5)))

    # ---------- 动作分发 ----------
    if action_id == "start_wave" or action_id == "wave_start":
        # 开始当前波次
        current_wave += 1
        enemy_scale = float(public_state.get("enemy_scale", 1.0))
        enemy_base_hp = int(float(public_state.get("enemy_base_hp", 30)))
        enemy_count = min(3 + current_wave, 8)

        enemy_list = []
        for i in range(enemy_count):
            hp = int(enemy_base_hp * enemy_scale * (1 + current_wave * 0.2))
            enemy_list.append({
                "id": f"enemy_{current_wave}_{i}",
                "name": _random_enemy_name(current_wave),
                "hp": hp,
                "max_hp": hp,
                "atk": 5 + current_wave * 2,
                "alive": True,
            })

        public_state["current_wave"] = current_wave
        public_state["enemy_list"] = enemy_list
        public_state["active_enemy_count"] = enemy_count
        public_state["current_status"] = "battle"
        session_update["phase"] = "battle"
        session_update["status"] = STATUS_ACTIVE
        session_update["round"] = int(float(session.get("round", 0))) + 1
        ui_update["phase_label"] = f"第{current_wave}波"
        ui_update["narration"] = f"第{current_wave}波来袭！{enemy_count}只敌人出现！"
        pending_narration = f"第{current_wave}波开始了，{enemy_count}只怪物咆哮着冲向你！"

    elif action_id == "attack" or action_id == "player_attack":
        # 玩家攻击
        player_atk = int(float(card.get("atk", 15)))
        target_id = str(params.get("target", ""))
        enemy_list = as_array(public_state.get("enemy_list", []))
        hit_count = 0

        for enemy in enemy_list:
            if enemy.get("alive") and (not target_id or enemy.get("id") == target_id):
                dmg = max(1, player_atk - int(float(enemy.get("def", 0)))
                           + random.randint(-3, 3))
                enemy["hp"] = max(0, enemy["hp"] - dmg)
                enemy["alive"] = enemy["hp"] > 0
                hit_count += 1
                log_entry = f"你攻击了{enemy['name']}，造成{dmg}点伤害"
                public_state["log"].append(log_entry)
                public_state["total_damage_dealt"] = float(public_state.get("total_damage_dealt", 0)) + dmg

        public_state["enemy_list"] = enemy_list
        alive_count = sum(1 for e in enemy_list if e.get("alive"))
        public_state["active_enemy_count"] = alive_count

        if alive_count == 0:
            public_state["current_status"] = "wave_clear"
            pending_narration = f"第{current_wave}波清理完成！获得准备时间。"
        else:
            ui_update["narration"] = f"你发动攻击，命中{hit_count}个目标！"
            pending_narration = f"你挥剑斩向敌人，刀光闪过！"

    elif action_id == "use_skill":
        # 使用技能
        skill_name = str(params.get("skill", ""))
        player_mp = int(float(card.get("mp", 0)))
        skill_costs = {"fire_ball": 20, "ice_strike": 15, "heal": 25, "power_strike": 10}
        cost = skill_costs.get(skill_name, 15)

        if player_mp < cost:
            return {"ok": False, "error": f"MP不足，需要{cost}点MP"}

        player_mp -= cost
        card["mp"] = player_mp
        player["parameterCardJson"] = card
        public_state["player_mp"] = player_mp

        enemy_list = as_array(public_state.get("enemy_list", []))
        aoe_dmg = int(float(card.get("atk", 15))) * 2 + random.randint(5, 15)
        for enemy in enemy_list:
            if enemy.get("alive"):
                enemy["hp"] = max(0, enemy["hp"] - aoe_dmg)
                enemy["alive"] = enemy["hp"] > 0

        public_state["enemy_list"] = enemy_list
        alive_count = sum(1 for e in enemy_list if e.get("alive"))
        public_state["active_enemy_count"] = alive_count
        pending_narration = f"你施放了{skill_name}，对全体敌人造成{aoe_dmg}点伤害！"

        if alive_count == 0:
            public_state["current_status"] = "wave_clear"

    elif action_id == "use_item":
        # 使用物品
        item_name = str(params.get("item", ""))
        inventory = as_array(state.get("inventory", []))
        item_match = None
        for item in inventory:
            if scalar_text(item.get("name", "")) == item_name or scalar_text(item.get("itemName", "")) == item_name:
                item_match = item
                break

        if not item_match:
            return {"ok": False, "error": f"物品 {item_name} 不在背包中"}

        # 移除物品
        state["inventory"] = [i for i in inventory
                             if scalar_text(i.get("name", "")) != item_name
                             and scalar_text(i.get("itemName", "")) != item_name]
        # 效果
        item_effects = {"血瓶": (("hp", 50),), "大血瓶": (("hp", 100),), "魔瓶": (("mp", 30),)}
        effects = item_effects.get(item_name, ())
        for key, val in effects:
            if key == "hp":
                card["hp"] = min(int(float(card.get("max_hp", 100))), int(float(card.get("hp", 0))) + val)
            elif key == "mp":
                card["mp"] = min(int(float(card.get("max_mp", 50))), int(float(card.get("mp", 0))) + val)
        player["parameterCardJson"] = card
        pending_narration = f"你使用了{item_name}，恢复了状态！"

    elif action_id == "enemy_turn":
        # 敌方回合（AI 编排时由后端触发）
        enemy_list = as_array(public_state.get("enemy_list", []))
        player_hp = int(float(card.get("hp", 100)))
        total_dmg = 0
        for enemy in enemy_list:
            if enemy.get("alive"):
                atk = int(float(enemy.get("atk", 10)))
                dmg = max(1, atk - int(float(card.get("def", 5))))
                player_hp = max(0, player_hp - dmg)
                total_dmg += dmg
                public_state["log"].append(f"{enemy['name']}攻击了你，造成{dmg}伤害")

        card["hp"] = player_hp
        player["parameterCardJson"] = card
        public_state["player_hp"] = player_hp
        public_state["total_damage_dealt"] = float(public_state.get("total_damage_dealt", 0))

        if player_hp <= 0:
            public_state["current_status"] = "defeat"
            session_update["status"] = STATUS_FINISHED
            pending_narration = "你倒在了荒野之中，挑战失败..."
        else:
            pending_narration = f"敌人发起反击，你承受了{total_dmg}点伤害！"

    elif action_id == "next_wave":
        # 下一波
        if current_wave >= wave_count:
            public_state["current_status"] = "victory"
            session_update["status"] = STATUS_SETTLING
            pending_narration = "所有波次已清理，你成功征服了野外生存挑战！"
        else:
            action_id = "wave_start"  # 重路由到 wave_start

    elif action_id == "exit_game" or action_id == "abort":
        # 退出游戏
        session_update["status"] = STATUS_ABORTED
        public_state["current_status"] = "aborted"
        pending_narration = "你选择了退出挑战。"

    else:
        return {"ok": False, "error": f"未知动作: {action_id}"}

    # 胜利检测
    if public_state.get("current_status") == "wave_clear" and current_wave >= wave_count:
        public_state["current_status"] = "victory"
        session_update["status"] = STATUS_SETTLING
        pending_narration = "完美通关！所有波次已清理！"

    # 更新 state_items（对齐 buildMiniGameUiStateItems）
    ui_update.setdefault("state_items", _build_state_items(public_state))
    ui_update.setdefault("narration", ui.get("narration", ""))
    if pending_narration:
        ui_update["narration"] = pending_narration

    # 构建 writeback（对齐 MiniGameController 的 writeback 机制）
    writeback_delta = {
        "player": player,
        "inventory": state.get("inventory", []),
    }

    # 奖励汇总（仅在 finish 时实际发放）
    if public_state.get("current_status") in ("victory", "defeat", "aborted"):
        total_kills = int(float(public_state.get("total_enemies_killed", 0)))
        total_dmg = float(public_state.get("total_damage_dealt", 0))
        reward_summary = {
            "exp": total_kills * 20 + current_wave * 10,
            "money": total_kills * 5 + current_wave * 3,
            "items": [],
            "level_ups": 0,
        }

    return {
        "ok": True,
        "step_result": {
            "messages": result_messages,
            "pending_narrative_plan": {
                "role": "narrator",
                "roleType": "narrator",
                "eventType": "mini_game_narration",
                "content": pending_narration,
            } if pending_narration else None,
            "reward_summary": reward_summary,
            "writeback": writeback_delta,
            "session_update": session_update,
            "ui_update": ui_update,
            "public_state_update": public_state,
        }
    }


def settle_game(
    state: Dict,
    result: str,  # "victory" | "defeat" | "aborted"
    reward_summary: Dict,
    ctx: Optional[Dict] = None,
) -> Dict:
    """
    游戏结束时发放奖励，更新角色参数卡。
    对齐 MiniGameController.ts 的奖励发放逻辑（normalizeMiniGameParameterCardProgress）

    参数：
        state          — 当前会话状态
        result          — 游戏结果
        reward_summary  — apply_action 返回的奖励摘要
        ctx            — 额外上下文

    返回：
        {
            "ok": True,
            "state_delta": { ... },   # 要写入 session.state 的变更
            "settlement_narration": "...",
        }
    """
    player = as_record(state.get("player", {}))
    card = as_record(player.get("parameterCardJson", {}))
    inventory = as_array(state.get("inventory", []))

    exp = int(float(reward_summary.get("exp", 0)))
    money = int(float(reward_summary.get("money", 0)))
    items = as_array(reward_summary.get("items", []))

    # 经验处理（对齐 normalizeMiniGameParameterCardProgress）
    current_exp = int(float(card.get("exp", 0)))
    current_level = max(1, int(float(card.get("level", 1))))
    current_next_level = int(float(card.get("next_level_exp", 100)))
    current_money = int(float(card.get("money", 0)))

    new_exp = current_exp + exp
    new_level = current_level
    new_next_level = current_next_level
    level_ups = 0

    while new_exp >= new_next_level:
        new_exp -= new_next_level
        new_level += 1
        new_next_level = max(100, new_level * 100)
        level_ups += 1

    # 发放奖励
    card["exp"] = new_exp
    card["level"] = new_level
    card["next_level_exp"] = new_next_level
    card["money"] = current_money + money

    # 升级时满血满蓝（对齐 resolveMiniGameFullResource）
    if level_ups > 0:
        max_hp = 100 + new_level * 10
        max_mp = 100 + new_level * 10
        card["hp"] = max_hp
        card["mp"] = max_mp
        card["max_hp"] = max_hp
        card["max_mp"] = max_mp

    # 添加物品到背包
    for item in items:
        item_name = scalar_text(item.get("name", ""))
        if item_name:
            inventory.append({
                "name": item_name,
                "kind": scalar_text(item.get("kind", "loot")),
                "rarity": scalar_text(item.get("rarity", "normal")),
            })

    # 去重物品
    seen = set()
    unique_inventory = []
    for it in inventory:
        name = scalar_text(it.get("name", ""))
        if name and name not in seen:
            seen.add(name)
            unique_inventory.append(it)
    inventory = unique_inventory

    player["parameterCardJson"] = card
    state["inventory"] = inventory

    # 叙事文本
    if result == "victory":
        narration = f"挑战胜利！获得 {exp} 经验、{money} 金币"
        if level_ups > 0:
            narration += f"，并升了 {level_ups} 级！"
        else:
            narration += "。"
    elif result == "defeat":
        narration = "挑战失败... 但你获得了宝贵的经验教训。"
    else:
        narration = "你退出了挑战，未获得奖励。"

    return {
        "ok": True,
        "state_delta": {
            "player": player,
            "inventory": inventory,
        },
        "settlement_narration": narration,
    }

def get_intent_prompt(
    current_state: Dict,
    available_actions: List[str],
    ctx: Optional[Dict] = None,
) -> str:
    """
    返回意图解析提示词片段。
    当用户在小游戏中输入文字时，AI 根据此提示词将文字映射为 action_id。
    对齐 MiniGameIntentService.ts 的 miniGameIntentSchema
    """
    action_lines = "\n".join(f"- `{aid}`: {label}" for aid, label in available_actions)

    return f"""
## {DISPLAY_NAME} 意图解析

当前状态：
- 波次：{current_state.get("current_wave", 0)}/{current_state.get("wave_count", 5)}
- HP：{current_state.get("player_hp", 0)}/{current_state.get("player_max_hp", 100)}
- MP：{current_state.get("player_mp", 0)}/{current_state.get("player_max_mp", 50)}
- 存活敌人：{current_state.get("active_enemy_count", 0)}

可用动作：
{action_lines}

解析规则：
- "攻击/打" → attack
- "用技能/施放" → use_skill（需要指定技能名）
- "用物品/喝血瓶" → use_item（需要指定物品名）
- "下一波/继续" → next_wave
- "退出/不玩了" → exit_game
""".strip()

def _random_enemy_name(wave: int) -> str:
    """根据波次生成随机敌人名称"""
    prefixes = ["野兽", "异兽", "怪物", "猛兽"]
    suffixes = ["狼", "熊", "鹰", "豹", "蟒", "蜥", "甲虫"]
    return f"{random.choice(prefixes)}{random.choice(suffixes)}{wave}型"


def _build_state_items(public_state: Dict) -> List[Dict]:
    """
    构建前端展示的状态条目。
    对齐 MiniGameController.ts 的 buildMiniGameUiStateItems（battle 部分）
    """
    items = [
        {"key": "当前波次", "value": f"{public_state.get('current_wave', 0)}/{public_state.get('wave_count', 5)}"},
        {"key": "生命值", "value": f"{public_state.get('player_hp', 0)}/{public_state.get('player_max_hp', 100)}"},
        {"key": "魔法值", "value": f"{public_state.get('player_mp', 0)}/{public_state.get('player_max_mp', 50)}"},
        {"key": "存活敌人", "value": f"{public_state.get('active_enemy_count', 0)}只"},
    ]
    if public_state.get("current_wave", 0) > 0:
        items.append({"key": "总伤害", "value": str(int(float(public_state.get("total_damage_dealt", 0))))})
    return items
