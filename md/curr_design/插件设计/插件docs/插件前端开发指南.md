# 插件前端开发指南

> 如何编写 `entry.js` 和 `ui/game.html`，与现有 ScenePlay.vue / VoiceInputPanel 完全对齐。
> 最后更新：2026-09-21

---

## 一、文件分工

| 文件 | 职责 | 谁执行 |
|------|------|--------|
| `manifest.json` | 插件元数据（类型、版本、描述、挂载方式） | 插件作者 |
| `entry.js` | 前端入口：选择面板、界面切换、事件总线 | 插件作者 |
| `ui/game.html` | Canvas 游戏或小游戏面板 HTML（内嵌 `<script>`） | 插件作者 |
| `ui/panel.html` | type=panel 时的浮层面板 | 插件作者 |
| 前端运行时 | 加载 entry.js、注入 ctx、渲染 UI、转发事件 | Toonflow 前端 |

---

## 二、manifest.json

```json
{
  "id": "toonflow-field-survival",
  "name": "野外生存",
  "version": "1.0.0",
  "author": "Your Name",
  "type": "minigame",
  "displayType": "小游戏",
  "gameType": "field_survival",
  "description": "2.5D ARPG 野外生存战斗小游戏",
  "thumbnail": "assets/thumbnail.png",
  "tags": ["rpg", "combat", "survival"],
  "minToonflowVersion": "1.0.0",
  "mount": {
    "mode": "overlay",
    "target": "#mini-game-panel"
  },
  "permissions": ["state:read", "state:write", "inventory:read", "inventory:write"]
}
```

**关键字段说明：**

- `type`: `"minigame"` 挂载小游戏面板；`"panel"` 独立浮层；`"standalone"` 独立页面
- `gameType`: 小游戏唯一标识，与后端 `MiniGameStateManager` 中的 `gameType` 对应
- `mount.mode`: `"overlay"` 叠加在会话界面；`"fullscreen"` 全屏替换；`"tab"` Tab 切换
- `mount.target`: 挂载点 CSS 选择器，默认 `#mini-game-panel`
- `permissions`: 声明需要的权限（前端沙箱用，后端以 entry.py 为准）

## 三、entry.js 完整结构

```javascript
// entry.js — 野外生存插件入口（完整注释参考）

/**
 * 野外生存 ARPG 小游戏插件入口
 *
 * 与现有 MiniGameController / ScenePlay.vue 对齐：
 * - 小游戏状态在 session.state.miniGame 中
 * - UI 渲染由 ScenePlay.vue 的 activeMiniGame computed 驱动
 * - 用户操作通过 miniGameAction API 发送到后端
 * - 后端 applyAction() 更新 public_state，AI 编排继续
 */

// ============================================================
// 阶段 1：声明插件元数据
// ============================================================
const plugin = {
  id: "toonflow-field-survival",
  name: "野外生存",
  version: "1.0.0",
  type: "minigame",
  gameType: "field_survival",

  // 游戏类型标签（与 MiniGameStateManager.gameType 对应）
  tags: ["survival", "combat", "rpg"],

  // 对齐 ScenePlay.vue 的 visibleStatuses
  visibleStatuses: new Set(["preparing", "active", "settling", "suspended"]),

  // 难度配置（选择面板用）
  difficulties: [
    { id: "normal",   label: "普通",    waveCount: 5,  enemyScale: 1.0 },
    { id: "hard",     label: "困难",    waveCount: 8,  enemyScale: 1.5 },
    { id: "nightmare",label: "噩梦",    waveCount: 10, enemyScale: 2.0 },
  ],
};

// ============================================================
// 阶段 2：前端沙箱 API 上下文（由 Toonflow 前端注入）
// ============================================================
let ctx = null; // Toonflow 运行时注入的 ctx 对象

/**
 * 初始化插件，接收 Toonflow 注入的上下文。
 * @param {Object} toonflowCtx - Toonflow 注入的沙箱 API
 */
function init(toonflowCtx) {
  ctx = toonflowCtx;
  console.log("[field-survival] 插件初始化, gameType:", ctx.gameType);
}

// ============================================================
// 阶段 3：插件 API（插件向 Toonflow 暴露的能力）
// ============================================================

/**
 * 插件元数据（前端显示用）
 */
function getMeta() {
  return {
    id: plugin.id,
    name: plugin.name,
    version: plugin.version,
    type: plugin.type,
    gameType: plugin.gameType,
  };
}

/**
 * 小游戏状态读取
 * 对应 MiniGameController.ts 中的 session.state.miniGame
 * @returns {Object|null} 当前小游戏状态
 */
function getMiniGameState() {
  if (!ctx || !ctx.state) return null;
  const root = (ctx.state.miniGame || {});
  const session = (root.session || {});
  const status = String(session.status || "");
  if (!plugin.visibleStatuses.has(status)) return null;
  return {
    gameType: String(session.game_type || session.gameType || plugin.gameType),
    status,
    phase: String(session.phase || ""),
    round: Number(session.round || 0),
    publicState: (session.public_state || {}),
    ui: (root.ui || {}),
    rulebook: (root.rulebook || {}),
    writeback: (root.writeback || {}),
  };
}

/**
 * 用户操作 → 发送到后端 miniGameAction API
 * 对应 ScenePlay.vue 中的 store.api.miniGameInventoryAction
 * @param {string} actionId - 动作 ID
 * @param {Object} params  - 动作参数
 */
async function dispatchAction(actionId, params = {}) {
  if (!ctx || !ctx.api) {
    console.error("[field-survival] ctx.api 不可用");
    return { ok: false, error: "api unavailable" };
  }
  try {
    // 对齐 useToonflowStore.ts 中的 miniGameInventoryAction 调用
    const result = await ctx.api.miniGameInventoryAction({
      action_id: actionId,
      params: params,
      game_type: plugin.gameType,
    });
    return { ok: true, result };
  } catch (err) {
    console.error("[field-survival] dispatchAction 失败:", err);
    return { ok: false, error: String(err) };
  }
}

/**
 * 读取角色参数卡（HP/MP/等级/技能/金币/经验/物品）
 * 对应 MiniGameController.ts 中的 createPlayerParameterCard()
 * @returns {Object} 参数卡 JSON
 */
function getPlayerCard() {
  if (!ctx || !ctx.state) return {};
  const player = (ctx.state.player || {});
  const card = (player.parameterCardJson || {});
  return {
    name: player.name || "用户",
    level: Number(card.level || 1),
    hp: Number(card.hp || 100),
    maxHp: Number(card.max_hp || 100),
    mp: Number(card.mp || 0),
    maxMp: Number(card.max_mp || 0),
    exp: Number(card.exp || 0),
    nextLevelExp: Number(card.next_level_exp || 100),
    money: Number(card.money || 0),
    skills: Array.isArray(card.skills) ? card.skills : [],
    items: Array.isArray(card.items) ? card.items : [],
    equipment: Array.isArray(card.equipment) ? card.equipment : [],
  };
}

/**
 * 读取物品栏
 * @returns {Array} 物品列表
 */
function getInventory() {
  if (!ctx || !ctx.state) return [];
  return Array.isArray(ctx.state.inventory) ? ctx.state.inventory : [];
}

/**
 * 写入变量（通过 writeback 机制，最终由 MiniGameController 合并）
 * 实际写入走 dispatchAction → 后端 writeback 合并
 * @param {string} path  - 路径，如 "player.parameterCardJson.hp"
 * @param {*}      value - 值
 */
function writeVariable(path, value) {
  if (!ctx) return;
  ctx.variables = ctx.variables || {};
  ctx.variables[path] = value;
}

/**
 * 读取变量
 * @param {string} path - 路径
 * @returns {*} 值
 */
function readVariable(path) {
  if (!ctx || !ctx.state) return undefined;
  // 支持点号路径
  return path.split(".").reduce((obj, key) => (obj && obj[key] !== undefined) ? obj[key] : undefined, ctx.state);
}

/**
 * 发送文本消息到故事会话（旁白/NPC 发言）
 * 实际走 ctx.api.addMessage → 后端 addMessage.ts
 * @param {string} role     - "narrator" | "npc" | "player"
 * @param {string} roleType - "narrator" | "npc" | "player"
 * @param {string} content  - 消息内容
 * @param {Object} meta    - 元数据（eventType 等）
 */
async function sendMessage(role, roleType, content, meta = {}) {
  if (!ctx || !ctx.api) return;
  try {
    await ctx.api.addMessage({
      role,
      roleType,
      content,
      meta,
    });
  } catch (err) {
    console.error("[field-survival] sendMessage 失败:", err);
  }
}

/**
 * 请求 AI 重新编排（重新触发叙事编排）
 * 相当于用户在输入框输入内容后发送
 * @param {string} message - 触发消息
 */
async function requestOrchestration(message) {
  if (!ctx || !ctx.api) return;
  try {
    await ctx.api.commitNarrativeTurn({ message });
  } catch (err) {
    console.error("[field-survival] requestOrchestration 失败:", err);
  }
}

/**
 * 插件间通信：广播事件
 * @param {string} event    - 事件名
 * @param {Object} payload  - 事件数据
 */
function emit(event, payload) {
  if (!ctx || !ctx.bus) return;
  ctx.bus.emit(event, payload);
}

/**
 * 插件间通信：监听事件
 * @param {string}   event   - 事件名
 * @param {Function} handler - 处理函数
 */
function on(event, handler) {
  if (!ctx || !ctx.bus) return;
  ctx.bus.on(event, handler);
}

/**
 * 渲染小游戏选择/说明面板（挂载到 #mini-game-panel）
 * 对应 ScenePlay.vue 中的 miniGamePanelExpanded / activeMiniGame
 *
 * @param {HTMLElement} container - 挂载容器

function renderSelectPanel(container) {
  container.innerHTML = `
    <div class="fs-panel">
      <div class="fs-header">
        <h2>🏕️ 野外生存</h2>
        <p>穿越危机四伏的荒野，击败来袭的敌人！</p>

        <button class="fs-btn fs-btn-ghost" id="fs-cancel">取消</button>

/**
 * 生成角色选项（从 ctx 获取当前会话中的角色）
 * 对齐 ScenePlay.vue 中的 worldRoles / npcs
 */
function buildRoleOptions() {
  if (!ctx || !ctx.state) return `<p class="fs-empty">无法获取角色列表</p>`;

  const roles = [];
  // 玩家角色
  const player = (ctx.state.player || {});
  if (player.name) {
    roles.push({
      id: player.id || "player",
      name: player.name,
      roleType: "player",
      avatar: player.avatarUrl || "",
      selected: true,
    });
  }
  // NPC 角色
  const npcs = (ctx.state.npcs || {});
  for (const [id, npc] of Object.entries(npcs)) {
    const n = (npc || {});
    if (n.name) {
      roles.push({
        id: String(id),
        name: n.name,
        roleType: n.roleType || "npc",
        avatar: n.avatarUrl || "",
        selected: false,
      });
    }
  }

  if (!roles.length) return `<p class="fs-empty">当前无角色可用</p>`;

  return roles.map(r => `
    <label class="fs-role-item ${r.selected ? "selected" : ""}" data-id="${r.id}">
      <input type="checkbox" name="fs-role" value="${r.id}"
             ${r.selected ? "checked" : ""}
             data-name="${r.name}"
             data-type="${r.roleType}">
      <span class="fs-role-name">${r.name}</span>
      <span class="fs-role-type">${r.roleType === "player" ? "主角" : "NPC"}</span>
    </label>
  `).join("");
}

/**
 * 绑定选择面板事件
 */
function bindSelectPanelEvents(container) {
  // 开始挑战
  container.querySelector("#fs-start")?.addEventListener("click", async () => {
    const selected = getSelectedRoles();
    if (!selected.playerId) {
      alert("请选择至少一个角色");
      return;
    }
    // 难度从选中的按钮读
    const diffBtn = container.querySelector(".fs-btn-diff.active");
    const difficulty = diffBtn ? diffBtn.dataset.difficulty : "normal";
    // 通知 Toonflow 开启小游戏
    await dispatchAction("start_game", {
      playerId: selected.playerId,
      npcIds: selected.npcIds,
      difficulty,
    });
  });

  // 观战模式
  container.querySelector("#fs-watch")?.addEventListener("click", async () => {
    await dispatchAction("watch_game", {});
  });

  // 取消
  container.querySelector("#fs-cancel")?.addEventListener("click", () => {
    emit("plugin:cancel", { pluginId: plugin.id });
  });

  // 难度按钮切换
  container.querySelectorAll(".fs-btn-diff").forEach(btn => {
    btn.addEventListener("click", () => {
      container.querySelectorAll(".fs-btn-diff").forEach(b => b.classList.remove("active"));
      btn.classList.add("active");
    });
  });

  // 角色多选
  container.querySelectorAll('input[name="fs-role"]').forEach(input => {
    input.addEventListener("change", () => {
      const label = input.closest(".fs-role-item");
      label.classList.toggle("selected", input.checked);
    });
  });
}

/**
 * 从 DOM 读取选中的角色
 */
function getSelectedRoles() {
  const checked = Array.from(
    document.querySelectorAll('input[name="fs-role"]:checked')
  );
  const playerId = checked.find(c => c.dataset.type === "player")?.value || "";
  const npcIds = checked.filter(c => c.dataset.type !== "player").map(c => c.value);
  return { playerId, npcIds };
}

/**
 * 渲染 Canvas 游戏界面（挂载到 #mini-game-canvas）
 * @param {HTMLElement} container - 挂载容器
 * @param {Object}       gameState - 初始游戏状态
 */
function renderGame(container, gameState) {
  // 完整 Canvas ARPG 实现见 ui/game.html
  // 这里加载游戏页面
  const iframe = document.createElement("iframe");
  iframe.src = ctx.resolvePluginUrl("ui/game.html");
  iframe.style.cssText = "width:100%;height:100%;border:none;";
  container.appendChild(iframe);

  // 向 iframe 注入 ctx
  iframe.addEventListener("load", () => {
    const iframeWindow = iframe.contentWindow;
    if (iframeWindow.__toonflow_plugin_init) {
      iframeWindow.__toonflow_plugin_init(ctx);
    }
  });
}

// 阶段 5：事件总线（与 Toonflow 前端对接）
// ============================================================

/**
 * 插件主入口（由 Toonflow 前端调用）
 * @param {string}              action - "init" | "render" | "action" | "close"
 * @param {Object}              data   - 传递的数据
 * @param {Object}              toonflowCtx - Toonflow 注入的上下文
 */
function pluginMain(action, data, toonflowCtx) {
  switch (action) {
    case "init":
      init(toonflowCtx);
      break;

    case "render":
      // data.target: DOM 容器
      // data.mode: "select" | "game" | "summary"
      if (data.mode === "select") {
        renderSelectPanel(data.target);
      } else if (data.mode === "game") {
        renderGame(data.target, data.gameState);
      } else if (data.mode === "summary") {
        renderMiniGamePanel(data.target, data.gameState);
      }
      break;

    case "action":
      // 处理来自 Toonflow 前端的命令
      handleToonflowAction(data);
      break;

    case "close":
      // 清理资源
      close();
      break;

    default:
      console.warn("[field-survival] 未知 action:", action);
  }
}

/**
 * 处理来自 Toonflow 前端的 action
 * @param {Object} data
 */
function handleToonflowAction(data) {
  const { type, payload } = data;
  switch (type) {
    case "game_state_update":
      // 游戏状态更新通知
      emit("game:state_update", payload);
      break;
    case "ai_narration":
      // AI 叙事文本（来自 public_state.narration）
      emit("game:narration", payload);
      break;
    case "enemy_turn":
      // 敌方回合
      emit("game:enemy_turn", payload);
      break;
    default:
      break;
  }
}

/**
 * 清理（关闭游戏时调用）
 */
function close() {
  ctx = null;
  emit("plugin:close", { pluginId: plugin.id });
}

/**
 * 向 Canvas 游戏注入 ctx 并启动游戏循环
 * @param {Object} gameCtx - Toonflow 上下文
 */
function startGameLoop(gameCtx) {
  ctx = gameCtx;

  const gs = {
    player: { x: 320, y: 400, hp: 100, maxHp: 100, mp: 50, maxMp: 50,
              speed: 3, atk: 15, def: 5, level: 1, exp: 0, nextLevelExp: 100,
              money: 0, dir: "down", isMoving: false, attackTimer: 0,
              skills: [], items: [] },
    enemies: [],
    projectiles: [],
    particles: [],
    drops: [],
    wave: 0,
    maxWave: 5,
    enemyScale: 1.0,
    keys: {},
    running: true,
  };

  // 绑定键盘事件
  bindControls(gs);

  // 开始游戏循环
  requestAnimationFrame(function loop() {
    if (!gs.running) return;
    update(gs);
    render(gs);
    requestAnimationFrame(loop);
  });
}
